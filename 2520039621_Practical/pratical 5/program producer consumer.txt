#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <stdlib.h>
#include <time.h>

int main()
{
    int pipefd[2];
    pid_t pid;
    int data[] = {10,20,30,40,50};
    int value;
    int n = 5;

    if(pipe(pipefd) == -1)
    {
        perror("pipe");
        return 1;
    }

    pid = fork();

    if(pid < 0)
    {
        perror("fork");
        return 1;
    }

    if(pid > 0)
    {
        close(pipefd[0]);

        clock_t start = clock();

        for(int i = 0; i < n; i++)
        {
            value = data[i];
            write(pipefd[1],&value,sizeof(value));
            printf("Producer produced: %d\n",value);
        }

        close(pipefd[1]);

        clock_t end = clock();

        double time_taken = ((double)(end-start))/CLOCKS_PER_SEC;

        printf("Communication time: %f seconds\n",time_taken);

        wait(NULL);
    }
    else
    {
        close(pipefd[1]);

        while(read(pipefd[0],&value,sizeof(value)) > 0)
        {
            printf("Consumer consumed: %d\n",value);
        }

        close(pipefd[0]);
        exit(0);
    }

    return 0;
}
